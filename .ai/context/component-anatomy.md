└── lib/
    ├── <kebab-name>.js                   ← Main class

## Component Types

### Container (Accordion, TreeView, ScrollSpy)

- Extends `HTMLElement` with `extendsElement = 'section'`
- Manages children (often `<details>`) with `querySelectorAll`
- Uses `MutationObserver` to react to changes
- `name` attribute for grouping
- Emits events on child interaction

```html
<section is="wcag-accordion" name="group-1">
  <details is="wcag-details">...</details>
</section>
```

### Form Control (Input, Checkbox, Radio, Switch, Textarea, Select)

- Extends the specific native element (`HTMLInputElement`, etc.)
- `aria-label` → visible `<label>` pattern with automatic wrapping:

```js
#init() {
  if (this.ariaLabel) {
    const fieldWrapper = DOM.wrapElement(this, { tag: 'span' });
    const label = DOM.wrapElement(fieldWrapper, { tag: 'label' });
    DOM.insertHTML(this.ariaLabel ?? 'aria-label N/A', label, 'prepend');
    this.removeAttribute('aria-label');
  }
}
```

Resulting DOM transformation:

```html
<!-- Before -->
<input is="wcag-input" type="text" aria-label="Username" />

<!-- After (generated by JS) -->
<label>
  Username
  <span>
    <input is="wcag-input" type="text" />
  </span>
</label>
```

- For checkbox/radio/switch: adds `sr-only` and pseudo-elements for custom appearance
- Handles events: `input`, `change`, `focus`, `blur`, `invalid`
- Supports `disabled`, `:disabled`, `aria-disabled="true"`

### Dialog (Dialog, Tooltip)

- Extends `HTMLDialogElement` or other
- Manages `open` attribute with `aria-expanded`
- Emits toggle events with `open`/`close` state

### Disclosure (Details)

- Extends `HTMLDetailsElement`
- Handles the `open` attribute natively
- Adds `aria-expanded`, `aria-controls`, GUID
- Chevron icon with `rotate` transition

## Attribute Callbacks (`<kebab>.attributes.js`)

Default export of a dictionary where:
- **Keys** are HTML attribute names (e.g. `'open'`, `'aria-label'`, `'name'`)
- **Values** are functions `(oldValue, newValue)` invoked when the attribute changes
- `this` is the component instance

```js
import { events } from '@wcag-ui/core';

export default {
  'aria-label': function (_oldValue, newValue) {
    if (this.label) {
      this.label.childNodes[0].textContent = newValue;
      events.dispatchComponentEvent.call(this, 'aria-label.change', { label: newValue });
    }
  },
  name: function () {
    this.update();
  },
};
```

Keys are automatically converted to `observedAttributes` and routing handlers:
- `'open'` → `handleOpenAttributeChanged(oldValue, newValue)`
- `'aria-label'` → `handleAriaLabelAttributeChanged(oldValue, newValue)`

## Event Handlers (`<kebab>.events.js`)

Default export of a dictionary where:
- **Keys** are DOM event names (e.g. `'click'`, `'input'`, `'change'`)
- **Values** are handler functions `(e)`
- `this` is the component instance

```js
import { events } from '@wcag-ui/core';

export default {
  input: function (e) {
    events.dispatchComponentEvent.call(this, 'input', { value: this.value }, e);
  },
  change: function (e) {
    events.dispatchComponentEvent.call(this, 'change', { value: this.value }, e);
  },
  focus: function (e) {
    events.dispatchComponentEvent.call(this, 'focus', { value: this.value }, e);
  },
  blur: function (e) {
    events.dispatchComponentEvent.call(this, 'blur', { value: this.value }, e);
  },
};
```

Listeners are registered automatically:
- `connectedCallback()` → `addEventListener` for each key
- `disconnectedCallback()` → `removeEventListener` for each key

You can also listen to custom events from child components:

```js
'wcag-details.toggle': function (_e) {
  events.dispatchComponentEvent.call(this, 'toggle', {});
}
```

## Component CSS

### `_core.css` — Base Structure

```css
/* 1. Token definition on :root */
:root {
  --wcag-<component>--color: var(--wcag-c--neutral-800);
  --wcag-<component>--background-color: var(--wcag-c--light);
  --wcag-<component>--border-color: var(--wcag-c--neutral-800);
  --wcag-<component>--border-radius: var(--wcag-r--s);
  --wcag-<component>--padding: var(--wcag-s--400);
}

/* 2. Base selector — uses intermediate variables */
[is="wcag-<component>"] {
  color: var(--wcag-<component>--color);
  background-color: var(--wcag-<component>--background-color);
  border: 1px solid var(--wcag-<component>--border-color);
  border-radius: var(--wcag-<component>--border-radius);
  padding: var(--wcag-<component>--padding);
}
```

### `<kebab>.css` — entry point

```css
@import "./_core.css" layer(wcag-ui.components.<kebab>);
```

## New Component Checklist

- [ ] `static extendsElement` set to the correct native element
- [ ] `static attributes` imported from `.attributes.js` file
- [ ] `static events` imported from `.events.js` file
- [ ] `static { componentDecorator(this); }` in static block
- [ ] `constructor()` calls `super()` then `this.#init()`
- [ ] `#init()` configures DOM, ARIA, wrapping
- [ ] CSS in `@layer wcag-ui.components.<kebab>`
- [ ] CSS tokens defined on `:root` with `--wcag-<kebab>--` prefix
- [ ] HTML demo in `src/components-<kebab>.html`
